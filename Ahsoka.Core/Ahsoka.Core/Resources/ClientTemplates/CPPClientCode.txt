#include "DefaultServiceClient.h"
#include "DefaultServiceMessages.hpp"
#include "AhsokaClientEndPoint.h"

using namespace std;
using namespace DefaultNameSpace;

namespace DefaultNameSpace
{
    typedef AhsokaClientEndPoint<DefaultTransportIdEnum> DefaultService_t;
	
	DefaultServiceClient::DefaultServiceClient():
       AhsokaServiceFramework::AhsokaEndPointWithCallback<DefaultTransportIdEnum>(new DefaultService_t(AhsokaConfiguration::GetServiceConfiguration(DefaultServiceName), DefaultServiceMessages()))
    {
        
    }

    DefaultServiceClient::DefaultServiceClient(ServiceConfiguration config):
	    AhsokaServiceFramework::AhsokaEndPointWithCallback<DefaultTransportIdEnum>(new DefaultService_t(config, DefaultServiceMessages()))
    {
        
    }

	void DefaultServiceClient::RegisterListener(MessageReceived_t callback)
    {
        return GetEndPoint<DefaultService_t>()->NotifyMessageReceived(callback);     
    }
