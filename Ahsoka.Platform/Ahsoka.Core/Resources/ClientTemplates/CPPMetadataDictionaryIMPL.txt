
    std::map<uint,std::map<int, CanPropertyInfo>>& CanModelMetadata::GetAllMetadata()
    {
        if (metadata.size() == 0)
            LoadMetadata();

        return metadata;
    }

    std::map<int, CanPropertyInfo>& CanModelMetadata::GetMetadata(uint canID) 
    { 
        if (metadata.size() == 0)
            LoadMetadata(); 

        return metadata[canID]; 
    }

    CanPropertyInfo& CanModelMetadata::GetPropertyInfo(uint canID, uint signal, bool isSignalEnum) 
    { 
        static CanPropertyInfo emptyInfo;
     
        if (metadata.size() == 0)
            LoadMetadata(); 

        if (isSignalEnum)
            return metadata[canID][signal]; 
        else
        {
            for(auto& it : metadata[canID])
                if (it.second.GetSignalId() == signal)
                    return it.second;
                
            return emptyInfo;
        }
    }
