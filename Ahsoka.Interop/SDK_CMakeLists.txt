cmake_minimum_required (VERSION 3.8)

# Add Proto If needed
if (EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/Ahsoka.Proto/CMakeLists.txt")
    message("Adding Proto Build: " ${CMAKE_CURRENT_SOURCE_DIR}/Ahsoka.Proto)
    add_subdirectory ("Ahsoka.Proto")
endif()

#Determine Platfrom
set(AhsokaPlatform $ENV{AhsokaPlatform})
message("Building for Platform: " ${AhsokaPlatform})

# Add ProtoBuf and zero mq as well as our Proto Compiled Outputs
find_library ( PROTO_LIB NAMES protobuf )
find_library ( ZMQ_LIB NAMES zmq)

#Create Variable for Service Headers
get_filename_component(Messages inc/messages/ ABSOLUTE)
get_filename_component(Services inc/services/ ABSOLUTE)
set(Headers ${Messages} ${Services})

#Create Variable for Library
get_filename_component(Library lib/libAhsoka.Core.so ABSOLUTE)

# Set Globals
set(AhsokaHeaders ${Headers} ${PROTO_HEADER_DIR} PARENT_SCOPE)
set(AhsokaLibs ${Library} zmq protobuf Ahsoka.Proto PARENT_SCOPE)

#Get Client Implementations
FILE(GLOB Clients inc/services/*.cpp)
set(AhsokaSources ${Clients} PARENT_SCOPE)

function(add_ahsoka target_name)

    ADD_CUSTOM_COMMAND(TARGET ${target_name}
            PRE_BUILD
            COMMAND echo Remove Outputs
            COMMAND ${CMAKE_COMMAND} -E remove_directory ${CMAKE_SOURCE_DIR}/Outputs/CommandLine
            COMMAND ${CMAKE_COMMAND} -E remove_directory ${CMAKE_SOURCE_DIR}/Outputs/.appinfo
            COMMAND ${CMAKE_COMMAND} -E remove ${CMAKE_SOURCE_DIR}/Outputs/libAhsoka.Core.so
    )

    ADD_CUSTOM_COMMAND(TARGET ${target_name}
            POST_BUILD
            COMMAND echo Create Command Line Folder
            COMMAND ${CMAKE_COMMAND} -E make_directory ${CMAKE_SOURCE_DIR}/Outputs/CommandLine
            COMMAND ${CMAKE_COMMAND} -E copy_directory ${CMAKE_SOURCE_DIR}/.platform_support/$ENV{AhsokaPlatform}_TargetSupport/CommandLine ${CMAKE_SOURCE_DIR}/Outputs/CommandLine
            COMMAND ${CMAKE_COMMAND} -E copy_directory ${CMAKE_SOURCE_DIR}/.platform_support/.appinfo ${CMAKE_SOURCE_DIR}/Outputs/.appinfo
            COMMAND ${CMAKE_COMMAND} -E copy ${CMAKE_SOURCE_DIR}/.platform_support/$ENV{AhsokaPlatform}_TargetSupport/SDK/lib/libAhsoka.Core.so ${CMAKE_SOURCE_DIR}/Outputs/libAhsoka.Core.so
    )

endfunction()
